\hmodel{Mean-field Ising (01/02) and Hawkes (09). $s_i$: the 1-min active or talk spin of agent $i$. VR: variance ratio. $g_{\rm eq}$: equal-time loop gain (H02's $\beta J_0 q$, H04's $K$, H05's $g$). $n_x$, $n_s$: Hawkes cross- and self-offspring per event. $w$: the share within one $\Delta=60$\,s bin (kernel time $\tau_k$). Dilution: per turn, $\bar k$ unread messages from $N_{\rm room}-1$ partners and a response budget $b$. $y_{pm}$: the gain of method $m$ in period $p$. Control parameter: $x_{\rm att}$.}

\hmath{\vspace{-8pt}\begin{gather*}
g_{\rm eq}=1-1/\mathrm{VR},\qquad x_{\rm att}=(N_{\rm room}-1)/(1+\bar k)\\
\mathrm{VR}=\operatorname{Var}\!\big(\textstyle\sum_i s_i\big)/\sum_i\operatorname{Var}s_i,\qquad n_x=b\,x_{\rm att}\\
\hat g_{\rm map}=\frac{2wn_x}{1+2w(n_x+n_s)},\quad w=1-\tfrac{\tau_k}{\Delta}\big(1-e^{-\Delta/\tau_k}\big)\\
y_{pm}=\alpha_m+c_m\,x_{{\rm att},p}+u_p+\varepsilon_{pm},\quad\text{collapse: } c_m>0\ \forall m
\end{gather*}\vspace{-10pt}}
